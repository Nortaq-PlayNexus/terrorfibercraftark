    Nitrado¶

    Async client for the Nitrado REST API (v1).

    Manages ARK ASE CrossPlay servers: list, restart, stop/start, and push Game.ini / GameUserSettings.ini content.

    class NitradoClient:
    """An asyncio-friendly wrapper around the Nitrado REST API.

    Uses :mod:`httpx` behind the scenes and exposes small, typed methods for
    the operations the toolkit needs. Requires a Nitrado API token.
    """

    def __init__(self, token: str, base: str = API_BASE, timeout: float = 20.0):
        if not token:
            raise ValueError("Nitrado API token is required")
        self._token = token
        self._base = base.rstrip("/")
        self._client: httpx.AsyncClient | None = None
        self._timeout = timeout

    @property
    def _http(self) -> httpx.AsyncClient:
        if self._client is None:
            self._client = httpx.AsyncClient(
                base_url=self._base,
                timeout=self._timeout,
                headers={
                    "Authorization": f"Bearer {self._token}",
                    "Accept": "application/json",
                },
            )
        return self._client

    async def _request(
        self,
        method: str,
        path: str,
        data: dict[str, str] | None = None,
    ) -> dict[str, object]:
        r = await self._http.request(method, path, data=data)
        if r.status_code in (401, 403):
            raise NitradoAuthError(f"{r.status_code} on {method} {path}")
        if r.status_code == 404:
            raise NitradoNotFoundError(f"{r.status_code} on {method} {path}")
        r.raise_for_status()
        try:
            payload = r.json()
        except json.JSONDecodeError:
            return {"raw": r.text}
        if not isinstance(payload, dict):
            raise NitradoError(f"Unexpected JSON shape from {method} {path}")
        return dict(payload)

    async def list_services(self) -> list[dict[str, object]]:
        r = await self._request("GET", "/services")
        data = r.get("data") or {}
        if isinstance(data, dict):
            services = data.get("services")
            if isinstance(services, list):
                return [dict(s) for s in services if isinstance(s, dict)]
        services = r.get("services")
        if isinstance(services, list):
            return [dict(s) for s in services if isinstance(s, dict)]
        return []

    async def gameserver(self, service_id: int | str, gs_id: int | str) -> dict[str, object]:
        r = await self._request("GET", f"/services/{service_id}/gameservers/{gs_id}")
        data = r.get("data")
        if isinstance(data, dict):
            gs = data.get("gameserver")
            if isinstance(gs, dict):
                return dict(gs)
        return r

    async def restart(
        self, service_id: int | str, gs_id: int | str, message: str = "Admin restart"
    ) -> dict[str, object]:
        return await self._request(
            "POST",
            f"/services/{service_id}/gameservers/{gs_id}/restart",
            data={"message": message},
        )

    async def stop(self, service_id: int | str, gs_id: int | str) -> dict[str, object]:
        return await self._request("POST", f"/services/{service_id}/gameservers/{gs_id}/stop")

    async def start(self, service_id: int | str, gs_id: int | str) -> dict[str, object]:
        return await self._request("POST", f"/services/{service_id}/gameservers/{gs_id}/start")

    async def get_settings(self, service_id: int | str, gs_id: int | str) -> dict[str, object]:
        return await self._request("GET", f"/services/{service_id}/gameservers/{gs_id}/settings")

    async def update_config(
        self,
        service_id: int | str,
        gs_id: int | str,
        game_ini: str | None = None,
        gameusersettings_ini: str | None = None,
    ) -> dict[str, object]:
        """Push ARK config file contents.

        Field names follow Nitrado's documented settings schema. Verify the
        exact field keys against the current Nitrado developer portal.
        """
        form: dict[str, str] = {}
        if game_ini is not None:
            form["settings[game.ini]"] = game_ini
        if gameusersettings_ini is not None:
            form["settings[gameusersettings.ini]"] = gameusersettings_ini
        if not form:
            raise NitradoError("No config provided to update_config")
        return await self._request(
            "POST",
            f"/services/{service_id}/gameservers/{gs_id}/settings",
            data=form,
        )

    async def aclose(self) -> None:
        if self._client is not None:
            await self._client.aclose()
            self._client = None
